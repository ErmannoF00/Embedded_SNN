cmake_minimum_required(VERSION 3.16)
project(research_snn_addon LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_executable(research_snn_protocol_cli protocol_cli.cpp)
target_compile_options(research_snn_protocol_cli PRIVATE -Wall -Wextra -Wpedantic -O2 -ffp-contract=off)
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/generated/weights.hpp")
  add_executable(research_snn_host host_cli.cpp)
  target_include_directories(research_snn_host PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/generated)
  target_compile_options(research_snn_host PRIVATE -Wall -Wextra -Wpedantic -O2 -ffp-contract=off)
else()
  message(STATUS "Weight header unavailable: building protocol parser only. Run research_upgrade.export_cpp for model CLI.")
endif()
